核心合约函数

// 1. 创建任务（任何人可调） createTask(address _subAgent, uint256 _amount, bytes32 _expectedHash, uint256 _deadline) → returns (uint256 taskId) // 2. 锁定资金（由 Owner 调用） lockFund(uint256 _taskId) external payable // 3. SubAgent 提交哈希（验证者角色控制） submitHash(uint256 _taskId, bytes32 _dataHash) // 4. 验证并自动放款（哈希匹配则转移 ETH） verifyHash(uint256 _taskId) nonReentrant returns (bool) // 5. 退款（仅 Owner，失败后调用） refund(uint256 _taskId) nonReentrant
